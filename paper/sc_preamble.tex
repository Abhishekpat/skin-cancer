% =====================================================================
%  sc_preamble.tex
%  Shared preamble for the Skin Cancer Detection IEEE paper (1-col & 2-col).

%  and \ifCLASSOPTIONtwocolumn are already defined by IEEEtran.
% =====================================================================

  hcaxis/.style={

% ---- Adaptive figure/chart widths ------------------------------------
% Charts should be compact in single-column mode and column-filling in
% two-column mode.  These macros are expanded at figure time when
% \linewidth already reflects the surrounding column/float width.
\ifCLASSOPTIONonecolumn
  \newcommand{\chartw}{0.62\linewidth}   % single (in-flow) chart
  \newcommand{\charth}{0.40\linewidth}
  \newcommand{\widechartw}{0.80\linewidth}
  \newcommand{\widecharth}{0.34\linewidth}
  \newcommand{\tallcharth}{0.52\linewidth}  % charts with many bars
  \newcommand{\barchartw}{0.52\linewidth}   % horizontal bar charts
\else
  \newcommand{\chartw}{0.98\linewidth}
  \newcommand{\charth}{0.66\linewidth}
  \newcommand{\widechartw}{0.44\linewidth}
  \newcommand{\widecharth}{0.30\linewidth}
  \newcommand{\tallcharth}{0.90\linewidth}
  \newcommand{\barchartw}{0.80\linewidth}
\fi

% ---- Wide-float helpers ----------------------------------------------
% Wide floats use an INLINE guard at the point of use in the body:
%   \ifCLASSOPTIONtwocolumn\begin{figure*}\else\begin{figure}\fi ...
%   \ifCLASSOPTIONtwocolumn\end{figure*}\else\end{figure}\fi
% The \ifCLASSOPTIONtwocolumn test is evaluated at run time and
% consistently selects the SAME environment for the matching begin/end,
% so LaTeX's environment checker never sees a mismatch.  (We deliberately
% do NOT wrap \begin/\end inside \newcommand bodies, which is brittle.)

% ---- Small semantic helpers ------------------------------------------
\newcommand{\code}[1]{\texttt{\small #1}}
\newcommand{\field}[1]{\texttt{\footnotesize #1}}
\newcolumntype{L}[1]{>{\raggedright\arraybackslash}p{#1}}
\newcolumntype{C}[1]{>{\centering\arraybackslash}p{#1}}
\newcolumntype{Y}{>{\raggedright\arraybackslash}X}

% Tighter list spacing to keep the paper dense.
\usepackage{enumitem}
